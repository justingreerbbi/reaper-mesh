cmake_minimum_required(VERSION 3.16)
project(reaper_mesh_host_protocol LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(reaper_protocol components/reaper_protocol/protocol.cpp)
target_include_directories(reaper_protocol PUBLIC components/reaper_protocol/include)
target_compile_options(reaper_protocol PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(reaper_device components/reaper_device/configuration.cpp components/reaper_device/session.cpp components/reaper_device/messages.cpp)
target_include_directories(reaper_device PUBLIC components/reaper_device/include)
target_link_libraries(reaper_device PUBLIC reaper_protocol)
target_compile_options(reaper_device PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(configuration_test tests/configuration_test.cpp)
target_link_libraries(configuration_test PRIVATE reaper_device)
add_executable(session_test tests/session_test.cpp)
target_link_libraries(session_test PRIVATE reaper_device)
add_executable(nvs_storage_test tests/nvs_storage_test.cpp heltec-v3/main/nvs_storage.cpp)
target_include_directories(nvs_storage_test PRIVATE tests/stubs heltec-v3/main)
target_link_libraries(nvs_storage_test PRIVATE reaper_device)
add_executable(codec_bridge tests/codec_bridge.cpp)
target_link_libraries(codec_bridge PRIVATE reaper_protocol)
enable_testing()
add_test(NAME protocol_selftest COMMAND codec_bridge selftest)
foreach(scenario commit retry conflict failure uncertain-old uncertain-new reboot eviction owners corruption record validation wire radio)
  add_test(NAME configuration_${scenario} COMMAND configuration_test ${scenario})
endforeach()
foreach(scenario sync identity reconnect storage version)
  add_test(NAME session_${scenario} COMMAND session_test ${scenario})
endforeach()
foreach(scenario init reads bounds commit uncertain)
  add_test(NAME nvs_${scenario} COMMAND nvs_storage_test ${scenario})
endforeach()
add_executable(messages_test tests/messages_test.cpp)
target_link_libraries(messages_test PRIVATE reaper_device)
foreach(scenario chunks expiry ack capacity validation)
  add_test(NAME messages_${scenario} COMMAND messages_test ${scenario})
endforeach()
add_executable(display_test tests/display_test.cpp heltec-v3/main/status_display.cpp)
target_include_directories(display_test PRIVATE tests/display_stubs heltec-v3/main components/reaper_device/include)
foreach(scenario updates init-failure tx-failure)
  add_test(NAME display_${scenario} COMMAND display_test ${scenario})
endforeach()
# This is a host-only codec target, not an ESP-IDF firmware image or flashing target.

target_include_directories(reaper_mesh PUBLIC components/reaper_mesh/include)
target_link_libraries(reaper_mesh PUBLIC reaper_protocol)
target_compile_options(reaper_mesh PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(radio_packet_test tests/radio_packet_test.cpp)
target_link_libraries(radio_packet_test PRIVATE reaper_mesh)
foreach(scenario roundtrip malformed)
  add_test(NAME radio_packet_${scenario} COMMAND radio_packet_test ${scenario})
endforeach()

add_executable(mesh_test tests/mesh_test.cpp)
target_link_libraries(mesh_test PRIVATE reaper_mesh)
foreach(scenario line loss asymmetric broadcast hops races bounds)
  add_test(NAME mesh_${scenario} COMMAND mesh_test ${scenario})
endforeach()

add_executable(regional_test tests/regional_test.cpp)
target_link_libraries(regional_test PRIVATE reaper_mesh)
add_test(NAME regional_airtime COMMAND regional_test)
